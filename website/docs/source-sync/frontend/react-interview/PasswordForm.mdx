# PasswordForm

~~~javascript
import { useState } from "react";

const PASSPORT_RULES = {
  USA: {
    regex: /^\d{2}[A-Za-z]{6}$/,
    placeholder: "12ABCDEF",
    message: "USA passport must be 2 digits followed by 6 letters.",
  },
  CA: {
    regex: /^[A-Za-z]{2}\d{6}$/,
    placeholder: "AB123456",
    message: "Canada passport must be 2 letters followed by 6 digits.",
  },
  UK: {
    regex: /^\d{9}$/,
    placeholder: "123456789",
    message: "UK passport must be 9 digits.",
  },
};

export default function PassportForm() {
  const [country, setCountry] = useState("USA");
  const [passportNumber, setPassportNumber] = useState("");
  const [touched, setTouched] = useState(false);

  const rule = PASSPORT_RULES[country];

  const normalizedPassport = passportNumber.trim();

  const isValid =
    normalizedPassport.length > 0 &&
    rule.regex.test(normalizedPassport);

  const showError = touched && !isValid;

  function handleCountryChange(event) {
    setCountry(event.target.value);
    setPassportNumber("");
    setTouched(false);
  }

  function handlePassportChange(event) {
    setPassportNumber(event.target.value);
  }

  function handleSubmit(event) {
    event.preventDefault();
    setTouched(true);

    if (!isValid) {
      return;
    }

    console.log({
      country,
      passportNumber: normalizedPassport,
    });
  }

  return (
    <form onSubmit={handleSubmit}>
      <div>
        <label htmlFor="country">Passport Country</label>

        <select
          id="country"
          value={country}
          onChange={handleCountryChange}
        >
          <option value="USA">USA</option>
          <option value="CA">Canada</option>
          <option value="UK">United Kingdom</option>
        </select>
      </div>

      <div>
        <label htmlFor="passport">Passport Number</label>

        <input
          id="passport"
          type="text"
          value={passportNumber}
          placeholder={rule.placeholder}
          onChange={handlePassportChange}
          onBlur={() => setTouched(true)}
          aria-invalid={showError}
          aria-describedby={
            showError ? "passport-error" : undefined
          }
        />

        {showError && (
          <p id="passport-error" role="alert">
            {rule.message}
          </p>
        )}
      </div>

      <button type="submit" disabled={!isValid}>
        Submit
      </button>
    </form>
  );
}
~~~
